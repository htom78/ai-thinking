def order_label(order):
    if order["status"] == "pending":
        return "待支付"
    if order["status"] == "awaiting_shipment":
        return "待发货"
    if order["status"] == "paid":
        return "已支付"
    if order["status"] == "canceled":
        return "已取消"
    return "未知"
