from sklearn.model_selection import cross_val_score
from sklearn.linear_model import LogisticRegression
from sklearn.ensemble import GradientBoostingClassifier,RandomForestClassifier
r=list(csv.DictReader(open('data/train.csv')))
cs=sorted({x['contract'] for x in r});rs=sorted({x['region'] for x in r})
X=np.array([[float(x['tenure_months']),float(x['monthly_fee']),float(x['support_calls'])]+[x['contract']==c for c in cs]+[x['region']==g for g in rs] for x in r],float);y=np.array([int(x['churned']) for x in r])
